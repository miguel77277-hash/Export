package com.pocketbuilder.app_wslenytv1792177359057;

import android.content.Context;
import android.content.SharedPreferences;

import java.io.BufferedReader;
import java.io.File;
import java.io.FileInputStream;
import java.io.FileOutputStream;
import java.io.InputStreamReader;
import java.io.OutputStreamWriter;
import java.util.ArrayList;
import java.util.Locale;

/** Guarda las listas del usuario (enlace, archivo o demostración) y su contenido en caché. */
final class PlaylistStore {

    static final String KIND_DEMO = "demo";
    static final String KIND_URL = "url";
    static final String KIND_FILE = "file";

    private static final String PREFS = "leny_tv_lists";
    private static final String KEY_LISTS = "saved_lists";
    private static final String KEY_ACTIVE = "active_id";
    private static final String SEP = "\u0001";
    private static final String ROW = "\n";

    static final class Playlist {
        long id;
        String name;
        String kind;
        String ref;
        int channels;
        long updated;

        Playlist(long id, String name, String kind, String ref) {
            this.id = id;
            this.name = name == null ? "" : name;
            this.kind = kind;
            this.ref = ref == null ? "" : ref;
        }

        boolean isDemo() {
            return KIND_DEMO.equals(kind);
        }

        String kindLabel() {
            if (isDemo()) return "Demostración (incluida)";
            if (KIND_FILE.equals(kind)) return "Archivo del teléfono";
            return "Enlace web";
        }

        String summary() {
            StringBuilder sb = new StringBuilder();
            sb.append(kindLabel());
            if (channels > 0) sb.append(" · ").append(channels).append(" entradas");
            if (updated > 0) sb.append(" · actualizada ").append(android.text.format.DateFormat.format("dd/MM HH:mm", updated));
            return sb.toString();
        }
    }

    private PlaylistStore() {
    }

    private static SharedPreferences prefs(Context c) {
        return c.getApplicationContext().getSharedPreferences(PREFS, Context.MODE_PRIVATE);
    }

    static ArrayList<Playlist> lists(Context c) {
        String raw = prefs(c).getString(KEY_LISTS, "");
        ArrayList<Playlist> out = new ArrayList<Playlist>();
        if (raw == null || raw.length() == 0) return out;
        String[] rows = raw.split(ROW);
        for (int i = 0; i < rows.length; i++) {
            String row = rows[i];
            if (row == null || row.trim().length() == 0) continue;
            String[] f = row.split(SEP, -1);
            if (f.length < 4) continue;
            try {
                Playlist p = new Playlist(Long.parseLong(f[0]), f[1], f[2], f[3]);
                if (f.length > 4) p.channels = parseInt(f[4]);
                if (f.length > 5) p.updated = parseLong(f[5]);
                out.add(p);
            } catch (Exception ignored) {
                // Fila dañada: se omite sin romper la lista completa.
            }
        }
        return out;
    }

    static void saveLists(Context c, ArrayList<Playlist> list) {
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < list.size(); i++) {
            Playlist p = list.get(i);
            if (i > 0) sb.append(ROW);
            sb.append(p.id).append(SEP)
                    .append(clean(p.name)).append(SEP)
                    .append(clean(p.kind)).append(SEP)
                    .append(clean(p.ref)).append(SEP)
                    .append(p.channels).append(SEP)
                    .append(p.updated);
        }
        prefs(c).edit().putString(KEY_LISTS, sb.toString()).apply();
    }

    static void add(Context c, Playlist p) {
        ArrayList<Playlist> all = lists(c);
        all.add(p);
        saveLists(c, all);
    }

    static void update(Context c, Playlist updated) {
        ArrayList<Playlist> all = lists(c);
        for (int i = 0; i < all.size(); i++) {
            if (all.get(i).id == updated.id) {
                all.set(i, updated);
                break;
            }
        }
        saveLists(c, all);
    }

    static void remove(Context c, long id) {
        ArrayList<Playlist> all = lists(c);
        for (int i = all.size() - 1; i >= 0; i--) {
            if (all.get(i).id == id) all.remove(i);
        }
        saveLists(c, all);
        deleteCache(c, id);
        if (activeId(c) == id) {
            prefs(c).edit().putLong(KEY_ACTIVE, all.isEmpty() ? -1 : all.get(0).id).apply();
        }
    }

    static long nextId(ArrayList<Playlist> list) {
        long max = 0;
        for (int i = 0; i < list.size(); i++) max = Math.max(max, list.get(i).id);
        return max + 1;
    }

    static long activeId(Context c) {
        return prefs(c).getLong(KEY_ACTIVE, -1);
    }

    static void setActiveId(Context c, long id) {
        prefs(c).edit().putLong(KEY_ACTIVE, id).apply();
    }

    static Playlist active(Context c) {
        long id = activeId(c);
        ArrayList<Playlist> all = lists(c);
        for (int i = 0; i < all.size(); i++) {
            if (all.get(i).id == id) return all.get(i);
        }
        return all.isEmpty() ? null : all.get(0);
    }

    static Playlist byId(Context c, long id) {
        ArrayList<Playlist> all = lists(c);
        for (int i = 0; i < all.size(); i++) {
            if (all.get(i).id == id) return all.get(i);
        }
        return null;
    }

    /** Crea la lista de demostración si el usuario todavía no tiene ninguna. */
    static void seedDemoIfEmpty(Context c) {
        ArrayList<Playlist> all = lists(c);
        if (all.size() > 0) return;
        Playlist demo = new Playlist(nextId(all), DemoData.NAME, KIND_DEMO, "demo");
        demo.updated = System.currentTimeMillis();
        all.add(demo);
        saveLists(c, all);
        setActiveId(c, demo.id);
    }

    /**
     * Lista ya guardada con la misma referencia (mismo enlace o mismo archivo), o
     * null. Sirve para no crear entradas repetidas cuando se añade dos veces la
     * misma lista: antes cada intento creaba una lista nueva con su copia del
     * archivo y el usuario acababa con varias filas iguales.
     */
    static Playlist findByRef(Context c, String ref) {
        if (ref == null || ref.trim().length() == 0) return null;
        String wanted = ref.trim();
        ArrayList<Playlist> all = lists(c);
        for (int i = 0; i < all.size(); i++) {
            Playlist p = all.get(i);
            if (p.isDemo()) continue;
            if (p.ref != null && p.ref.trim().equals(wanted)) return p;
        }
        return null;
    }

    /** Devuelve la lista ya guardada con ese enlace o crea una nueva. */
    static Playlist createUrlPlaylist(Context c, String name, String url) {
        Playlist existing = findByRef(c, url);
        if (existing != null) {
            if (name != null && name.length() > 0) existing.name = name;
            existing.updated = System.currentTimeMillis();
            update(c, existing);
            return existing;
        }
        ArrayList<Playlist> all = lists(c);
        Playlist p = new Playlist(nextId(all), name, KIND_URL, url);
        p.updated = System.currentTimeMillis();
        all.add(p);
        saveLists(c, all);
        return p;
    }

    /** Devuelve la lista ya guardada con ese archivo o crea una nueva. */
    static Playlist createFilePlaylist(Context c, String name, String uri) {
        Playlist existing = findByRef(c, uri);
        if (existing != null) {
            if (name != null && name.length() > 0) existing.name = name;
            existing.updated = System.currentTimeMillis();
            update(c, existing);
            return existing;
        }
        ArrayList<Playlist> all = lists(c);
        Playlist p = new Playlist(nextId(all), name, KIND_FILE, uri);
        p.updated = System.currentTimeMillis();
        all.add(p);
        saveLists(c, all);
        return p;
    }

    /** Crea una entrada nueva para la lista de demostración incluida. */
    static Playlist createDemoPlaylist(Context c) {
        ArrayList<Playlist> all = lists(c);
        Playlist p = new Playlist(nextId(all), DemoData.NAME, KIND_DEMO, "demo");
        p.updated = System.currentTimeMillis();
        all.add(p);
        saveLists(c, all);
        return p;
    }

    // ---- Caché del contenido de la lista ------------------------------------

    private static File cacheFile(Context c, long id) {
        return new File(c.getFilesDir(), "list_" + id + ".m3u");
    }

    /** Contenido guardado de la lista, o null si todavía no hay nada copiado. */
    static String readCache(Context c, Playlist p) {
        if (p == null) return null;
        if (p.isDemo()) return DemoData.text();
        File f = cacheFile(c, p.id);
        if (!f.exists()) return null;
        try {
            FileInputStream in = new FileInputStream(f);
            BufferedReader reader = new BufferedReader(new InputStreamReader(in, "UTF-8"));
            StringBuilder sb = new StringBuilder();
            String line;
            while ((line = reader.readLine()) != null) sb.append(line).append('\n');
            reader.close();
            return sb.length() == 0 ? null : sb.toString();
        } catch (Exception e) {
            return null;
        }
    }

    static void writeCache(Context c, Playlist p, String text) {
        if (p == null || text == null) return;
        try {
            FileOutputStream out = new FileOutputStream(cacheFile(c, p.id));
            OutputStreamWriter writer = new OutputStreamWriter(out, "UTF-8");
            writer.write(text);
            writer.close();
        } catch (Exception ignored) {
            // Sin caché la app sigue funcionando con la descarga en vivo.
        }
    }

    static void deleteCache(Context c, long id) {
        File f = cacheFile(c, id);
        if (f.exists()) f.delete();
    }

    static String nameFromUrl(String url) {
        if (url == null || url.length() == 0) return "Lista";
        try {
            java.net.URL u = new java.net.URL(url);
            String host = u.getHost() == null ? url : u.getHost();
            String path = u.getPath() == null ? "" : u.getPath();
            int slash = path.lastIndexOf('/');
            String file = slash >= 0 ? path.substring(slash + 1) : path;
            if (file.length() > 0) return host + " · " + file;
            return host;
        } catch (Exception e) {
            return url.length() > 40 ? url.substring(0, 40) : url;
        }
    }

    private static String clean(String v) {
        if (v == null) return "";
        return v.replace(SEP, " ").replace("\n", " ").replace("\r", " ").trim();
    }

    private static int parseInt(String v) {
        try {
            return Integer.parseInt(v.trim());
        } catch (Exception e) {
            return 0;
        }
    }

    private static long parseLong(String v) {
        try {
            return Long.parseLong(v.trim());
        } catch (Exception e) {
            return 0L;
        }
    }

    static String displayKind(String kind) {
        return kind == null ? "" : kind.toUpperCase(Locale.US);
    }
}

package com.pocketbuilder.app_wslenytv1792177359057;

import android.content.Context;
import android.graphics.Canvas;
import android.graphics.ColorFilter;
import android.graphics.Paint;
import android.graphics.PixelFormat;
import android.graphics.Rect;
import android.graphics.RectF;
import android.graphics.Typeface;
import android.graphics.drawable.Drawable;
import android.os.Handler;
import android.os.Looper;
import android.view.Gravity;
import android.view.View;
import android.widget.LinearLayout;

import java.text.SimpleDateFormat;
import java.util.Date;
import java.util.Locale;

/**
 * Barra superior de la pantalla general, copiada de la referencia:
 *
 *   NOMBRE  [teclado]  fecha  [reloj] hora  ......  [lupa] [refrescar] [engranaje] [apagado]
 *
 * Cada botón hace lo mismo que el botón equivalente que ya existía en la app:
 * teclado = escribir un enlace a mano, lupa = buscar, refrescar = volver a cargar
 * la lista, engranaje = AJUSTES y apagado = cerrar la app.
 *
 * REVISIÓN 50: los iconos estaban un poco por encima de la línea del texto. La
 * causa era que se dibujaban como "icono de un TextView": un TextView vacío
 * reserva además la altura de una línea de texto y por eso el dibujo subía unos
 * píxeles. Ahora cada icono es una vista propia (IconView) que se dibuja
 * exactamente en el centro de su cuadro, así queda a la altura de la fecha y de
 * la hora que tiene al lado.
 */
final class RogBar {

    /** Lo que hace cada botón; lo decide la pantalla principal. */
    interface Host {
        void onKeyboard();

        void onSearch();

        void onRefresh();

        void onSettings();

        void onPower();
    }

    private RogBar() {
    }

    static View create(Context ctx, String name, String version, final Host host) {
        LinearLayout bar = new LinearLayout(ctx);
        bar.setOrientation(LinearLayout.HORIZONTAL);
        bar.setGravity(Gravity.CENTER_VERTICAL);
        bar.setBackground(RogStyle.bar(ctx));
        bar.setPadding(Ui.dpi(ctx, 10), Ui.dpi(ctx, 4), Ui.dpi(ctx, 8), Ui.dpi(ctx, 4));

        TextView brand = Ui.text(ctx, name, 15f, RogStyle.TEXT, true);
        brand.setTypeface(Typeface.DEFAULT_BOLD);
        brand.setSingleLine(true);
        brand.setLetterSpacing(0.02f);
        bar.addView(brand, wrap());

        TextView ver = Ui.text(ctx, version, 9.5f, RogStyle.TEXT_DIM, false);
        ver.setSingleLine(true);
        LinearLayout.LayoutParams verParams = wrap();
        verParams.leftMargin = Ui.dpi(ctx, 4);
        bar.addView(ver, verParams);

        LinearLayout.LayoutParams gap = wrap();
        gap.leftMargin = Ui.dpi(ctx, 10);
        bar.addView(iconButton(ctx, new KeyboardIcon(0xFFF3F1F1, Ui.dp(ctx, 24f)),
                "Escribir un enlace a mano", new View.OnClickListener() {
                    @Override
                    public void onClick(View v) {
                        host.onKeyboard();
                    }
                }), gap);

        TextView date = new ClockText(ctx, "dd-MM-yyyy", 10.5f, RogStyle.DATE, false);
        LinearLayout.LayoutParams dateParams = wrap();
        dateParams.leftMargin = Ui.dpi(ctx, 12);
        bar.addView(date, dateParams);

        LinearLayout.LayoutParams clockGap = wrap();
        clockGap.leftMargin = Ui.dpi(ctx, 10);
        bar.addView(icon(ctx, new ClockIcon(0xFFF3F1F1, Ui.dp(ctx, 24f)), "Reloj"), clockGap);

        TextView time = new ClockText(ctx, "HH:mm:ss", 12f, RogStyle.TEXT, true);
        LinearLayout.LayoutParams timeParams = wrap();
        timeParams.leftMargin = Ui.dpi(ctx, 6);
        bar.addView(time, timeParams);

        View spacer = new View(ctx);
        bar.addView(spacer, new LinearLayout.LayoutParams(0, 1, 1f));

        bar.addView(iconButton(ctx, Icons.search(ctx, 0xFFF3F1F1),
                "Buscar un canal", new View.OnClickListener() {
                    @Override
                    public void onClick(View v) {
                        host.onSearch();
                    }
                }), wrap());
        bar.addView(iconButton(ctx, Icons.refresh(ctx, 0xFFF3F1F1),
                "Volver a cargar la lista", new View.OnClickListener() {
                    @Override
                    public void onClick(View v) {
                        host.onRefresh();
                    }
                }), wrap());
        bar.addView(iconButton(ctx, Icons.gear(ctx, RogStyle.GREEN),
                "Ajustes", new View.OnClickListener() {
                    @Override
                    public void onClick(View v) {
                        host.onSettings();
                    }
                }), wrap());
        bar.addView(iconButton(ctx, Icons.power(ctx, RogStyle.POWER),
                "Cerrar LENY TV", new View.OnClickListener() {
                    @Override
                    public void onClick(View v) {
                        host.onPower();
                    }
                }), wrap());

        return bar;
    }

    private static LinearLayout.LayoutParams wrap() {
        return new LinearLayout.LayoutParams(LinearLayout.LayoutParams.WRAP_CONTENT,
                LinearLayout.LayoutParams.WRAP_CONTENT);
    }

    private static View iconButton(Context ctx, Drawable icon, String label,
                                   View.OnClickListener click) {
        View button = icon(ctx, icon, label);
        button.setClickable(true);
        button.setFocusable(true);
        button.setOnClickListener(click);
        button.setBackground(Ui.buttonBg(ctx, 0x00000000, 0x33FFFFFF, 0x00000000, 4f));
        return button;
    }

    private static View icon(Context ctx, Drawable icon, String label) {
        IconView view = new IconView(ctx, icon, 38f);
        view.setContentDescription(label);
        return view;
    }

    /**
     * Cuadro cuadrado que dibuja su icono en el centro exacto. Sustituye al
     * "icono de TextView" que quedaba un poco alto respecto al texto de al lado.
     */
    private static final class IconView extends View {
        private final Drawable icon;
        private final float boxDp;

        IconView(Context ctx, Drawable icon, float boxDp) {
            super(ctx);
            this.icon = icon;
            this.boxDp = boxDp;
        }

        @Override
        protected void onMeasure(int widthMeasureSpec, int heightMeasureSpec) {
            int size = Ui.dpi(getContext(), boxDp);
            setMeasuredDimension(size, size);
        }

        @Override
        protected void onDraw(Canvas canvas) {
            if (icon == null) return;
            int width = getWidth();
            int height = getHeight();
            int iconSize = Ui.dpi(getContext(), boxDp * 0.52f);
            int left = (width - iconSize) / 2;
            int top = (height - iconSize) / 2;
            icon.setBounds(left, top, left + iconSize, top + iconSize);
            icon.draw(canvas);
        }
    }

    /** Texto que se actualiza solo cada segundo (fecha y hora de la barra). */
    private static final class ClockText extends TextView {

        private final Handler handler = new Handler(Looper.getMainLooper());
        private final SimpleDateFormat format;
        private final Runnable tick = new Runnable() {
            @Override
            public void run() {
                try {
                    setText(format.format(new Date()));
                } catch (Throwable ignored) {
                }
                handler.postDelayed(this, 1000);
            }
        };

        ClockText(Context ctx, String pattern, float sizeSp, int color, boolean bold) {
            super(ctx);
            format = new SimpleDateFormat(pattern, Locale.US);
            setText(format.format(new Date()));
            setTextSize(sizeSp);
            setTextColor(color);
            if (bold) setTypeface(Typeface.DEFAULT_BOLD);
            setSingleLine(true);
        }

        @Override
        protected void onAttachedToWindow() {
            super.onAttachedToWindow();
            handler.removeCallbacks(tick);
            handler.post(tick);
        }

        @Override
        protected void onDetachedFromWindow() {
            handler.removeCallbacks(tick);
            super.onDetachedFromWindow();
        }
    }

    private static abstract class BaseIcon extends Drawable {
        final Paint paint = new Paint(Paint.ANTI_ALIAS_FLAG);
        private final float size;
        private int alpha = 255;

        BaseIcon(int color, float size) {
            this.size = size;
            paint.setColor(color);
            paint.setStyle(Paint.Style.STROKE);
            paint.setStrokeCap(Paint.Cap.ROUND);
            paint.setStrokeJoin(Paint.Join.ROUND);
        }

        @Override
        public void setAlpha(int value) {
            alpha = value;
            paint.setAlpha(value);
        }

        @Override
        public void setColorFilter(ColorFilter cf) {
            paint.setColorFilter(cf);
        }

        @Override
        public int getOpacity() {
            return PixelFormat.TRANSLUCENT;
        }

        @Override
        public int getIntrinsicWidth() {
            return Math.round(size);
        }

        @Override
        public int getIntrinsicHeight() {
            return Math.round(size);
        }

        @Override
        public void draw(Canvas canvas) {
            Rect bounds = getBounds();
            float w = bounds.width();
            float h = bounds.height();
            float s = Math.min(w, h);
            if (s <= 0) return;
            canvas.save();
            canvas.translate(bounds.left + (w - s) / 2f, bounds.top + (h - s) / 2f);
            paint.setAlpha(alpha);
            paint.setStrokeWidth(Math.max(1f, s * 0.085f));
            drawIcon(canvas, s);
            canvas.restore();
        }

        abstract void drawIcon(Canvas canvas, float s);
    }

    /** Tecla: teclado con teclas y barra espaciadora. */
    private static final class KeyboardIcon extends BaseIcon {
        KeyboardIcon(int color, float size) {
            super(color, size);
        }

        @Override
        void drawIcon(Canvas canvas, float s) {
            paint.setStyle(Paint.Style.STROKE);
            canvas.drawRoundRect(new RectF(s * 0.06f, s * 0.26f, s * 0.94f, s * 0.74f),
                    s * 0.10f, s * 0.10f, paint);
            paint.setStyle(Paint.Style.FILL);
            for (int row = 0; row < 2; row++) {
                for (int i = 0; i < 4; i++) {
                    canvas.drawCircle(s * (0.24f + i * 0.17f), s * (0.40f + row * 0.14f),
                            s * 0.033f, paint);
                }
            }
            paint.setStyle(Paint.Style.STROKE);
            canvas.drawLine(s * 0.30f, s * 0.66f, s * 0.70f, s * 0.66f, paint);
        }
    }

    /** Reloj: esfera con agujas. */
    private static final class ClockIcon extends BaseIcon {
        ClockIcon(int color, float size) {
            super(color, size);
        }

        @Override
        void drawIcon(Canvas canvas, float s) {
            paint.setStyle(Paint.Style.STROKE);
            canvas.drawCircle(s * 0.5f, s * 0.5f, s * 0.38f, paint);
            canvas.drawLine(s * 0.5f, s * 0.5f, s * 0.5f, s * 0.26f, paint);
            canvas.drawLine(s * 0.5f, s * 0.5f, s * 0.70f, s * 0.58f, paint);
        }
    }
}
