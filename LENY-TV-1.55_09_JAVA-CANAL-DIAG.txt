package com.pocketbuilder.app_wslenytv1792177359057;

import java.util.Map;
import java.util.TreeMap;

/**
 * Una entrada de una lista IPTV o un enlace de vídeo suelto.
 *
 * Además de su nombre, su logo y su categoría, guarda TODO lo que la lista dice
 * de ese canal en concreto:
 *
 *  - Las cabeceras HTTP de su bloque (#EXTVLCOPT, #KODIPROP, #EXTHTTP y los
 *    parámetros detrás de "|": User-Agent, Referer, Origin, Cookie y otras).
 *  - De qué directiva salió cada cabecera, para poder explicarlo en el
 *    diagnóstico.
 *  - Las directivas de licencia (inputstream.adaptive.license_type,
 *    license_key y stream_headers). Se guardan y se identifican, pero NO se usan
 *    para descifrar nada: el motor DASH de LENY TV no admite DRM y aquí no se
 *    inventa compatibilidad con ClearKey ni con ningún otro sistema.
 *
 * La clave de licencia nunca se muestra en pantalla ni en el diagnóstico.
 */
final class Channel {
    final String name;
    final String url;
    final String logo;
    /**
     * Apartado del canal. Es el único dato que puede cambiar después de leer la
     * lista: si una entrada viene sin apartado pero es el mismo canal que otra
     * que sí lo tiene, se coloca en ese apartado (ver
     * {@link #adoptCategory(String)}).
     */
    String category;
    /**
     * Direcciones ALTERNATIVAS del mismo canal (REVISIÓN 53). Muchas listas
     * ponen varias direcciones detrás de un solo #EXTINF: la misma emisión en
     * otro servidor, una «opción 2», un «index.m3u8» o el bloque de Kodi. Antes
     * cada una se mostraba como un canal nuevo y sin apartado, y por eso salía un
     * montón de entradas sueltas en «Sin categoría» con nombres como
     * «index.m3u8» o «3420». Ahora se guardan aquí y NO se muestran aparte.
     */
    final java.util.ArrayList<String> alternatives = new java.util.ArrayList<String>();
    /** Cabeceras propias del canal (vacío si la lista no indica ninguna). */
    final Map<String, String> headers;
    /** De dónde salió cada cabecera: "#EXTVLCOPT:http-user-agent", "después de \"|\""... */
    final Map<String, String> headerSources;
    /** User-Agent del canal (comodidad; ya está dentro de {@link #headers}). */
    final String userAgent;
    /** Referer del canal (comodidad; ya está dentro de {@link #headers}). */
    final String referer;
    /** Sistema de DRM declarado por la lista (vacío si no declara ninguno). */
    final String licenseType;
    /** Clave declarada por la lista. Se guarda, pero NUNCA se muestra. */
    final String licenseKey;
    /** Cabeceras declaradas para la licencia (inputstream.adaptive.stream_headers). */
    final String licenseStreamHeaders;

    Channel(String name, String url, String logo, String category) {
        this(name, url, logo, category, null, null, "", "", "");
    }

    Channel(String name, String url, String logo, String category,
            String userAgent, String referer) {
        this(name, url, logo, category, single(userAgent, referer), null, "", "", "");
    }

    Channel(String name, String url, String logo, String category,
            Map<String, String> headers, Map<String, String> headerSources,
            String licenseType, String licenseKey, String licenseStreamHeaders) {
        this.name = name == null ? "" : name.trim();
        this.url = url == null ? "" : url.trim();
        this.logo = logo == null ? "" : logo.trim();
        this.category = cleanCategory(category);
        this.headers = Headers.copy(headers);
        this.headerSources = headerSources == null
                ? new TreeMap<String, String>() : new TreeMap<String, String>(headerSources);
        String ua = this.headers.get("User-Agent");
        String ref = this.headers.get("Referer");
        this.userAgent = ua == null ? "" : ua;
        this.referer = ref == null ? "" : ref;
        this.licenseType = licenseType == null ? "" : licenseType.trim();
        this.licenseKey = licenseKey == null ? "" : licenseKey.trim();
        this.licenseStreamHeaders = licenseStreamHeaders == null ? "" : licenseStreamHeaders.trim();
    }

    /**
     * Nombre del apartado tal y como se guarda: sin los adornos que muchas listas
     * ponen delante y detrás ("### 24/7 ###", "== CINE ==", "· DEPORTES ·") y con
     * los espacios de sobra juntos.
     *
     * Antes esos adornos se guardaban tal cual, así que la lista traía DOS
     * apartados para lo mismo: "### 24/7 ###" y "24/7". Ahora los dos se guardan
     * como "24/7" y en la columna de apartados sale una sola fila con todos sus
     * canales.
     */
    private static String cleanCategory(String raw) {
        if (raw == null) return M3uParser.NO_CATEGORY;
        String text = raw.trim();
        while (text.length() > 0 && "#*·|_=~- ".indexOf(text.charAt(0)) >= 0) {
            text = text.substring(1).trim();
        }
        while (text.length() > 0 && "#*·|_=~- ".indexOf(text.charAt(text.length() - 1)) >= 0) {
            text = text.substring(0, text.length() - 1).trim();
        }
        StringBuilder sb = new StringBuilder(text.length());
        boolean space = false;
        for (int i = 0; i < text.length(); i++) {
            char c = text.charAt(i);
            if (c == ' ' || c == '\t') {
                if (!space) sb.append(' ');
                space = true;
            } else {
                sb.append(c);
                space = false;
            }
        }
        String out = sb.toString().trim();
        return out.length() == 0 ? M3uParser.NO_CATEGORY : out;
    }

    private static Map<String, String> single(String userAgent, String referer) {
        Map<String, String> map = new TreeMap<String, String>();
        Headers.put(map, "User-Agent", userAgent);
        Headers.put(map, "Referer", referer);
        return map;
    }

    /**
     * Guarda una dirección alternativa del mismo canal.
     * Devuelve false si esa dirección ya estaba (la principal o una repetida).
     */
    boolean addAlternative(String alternativeUrl) {
        if (alternativeUrl == null) return false;
        String clean = alternativeUrl.trim();
        if (clean.length() == 0) return false;
        if (clean.equals(url)) return false;
        if (alternatives.contains(clean)) return false;
        alternatives.add(clean);
        return true;
    }

    /** true si este canal trae direcciones alternativas. */
    boolean hasAlternatives() {
        return !alternatives.isEmpty();
    }

    /** Direcciones del canal, empezando por la principal. */
    java.util.ArrayList<String> allUrls() {
        java.util.ArrayList<String> out = new java.util.ArrayList<String>();
        if (url.length() > 0) out.add(url);
        out.addAll(alternatives);
        return out;
    }

    /**
     * Coloca el canal en su apartado cuando la lista no lo indicaba pero otro
     * canal con el mismo nombre sí lo traía. No cambia un apartado que ya tenga.
     */
    void adoptCategory(String value) {
        if (value == null || value.trim().length() == 0) return;
        if (!M3uParser.NO_CATEGORY.equals(category)) return;
        category = cleanCategory(value);
    }

    boolean hasOwnHeaders() {
        return !headers.isEmpty();
    }

    /** true si la lista declara protección de contenido para este canal. */
    boolean hasLicense() {
        return licenseType.length() > 0 || licenseKey.length() > 0;
    }

    /** Nombres de las cabeceras del canal, para el diagnóstico. */
    String headerNames() {
        return Headers.names(headers);
    }

    /**
     * Resumen del bloque del canal: qué cabeceras trae y de qué directiva salió
     * cada una. No incluye valores, así que también sirve para pegarlo.
     */
    String blockSummary() {
        if (headers.isEmpty()) return "(la lista no indica cabeceras para este canal)";
        StringBuilder sb = new StringBuilder();
        for (Map.Entry<String, String> entry : headers.entrySet()) {
            if (entry == null || entry.getKey() == null) continue;
            if (sb.length() > 0) sb.append(" · ");
            sb.append(entry.getKey());
            String source = headerSources.get(entry.getKey());
            sb.append(" ← ").append(source == null || source.length() == 0 ? "lista" : source);
        }
        return sb.toString();
    }

    /**
     * Resumen de la protección de contenido declarada. Dice qué sistema indica la
     * lista y que el motor NO descifra contenido protegido. Nunca muestra la clave
     * ni afirma que ClearKey funcione.
     */
    String licenseSummary() {
        if (!hasLicense()) return "";
        StringBuilder sb = new StringBuilder();
        sb.append("la lista declara DRM");
        if (licenseType.length() > 0) sb.append(" (license_type=").append(licenseType).append(')');
        if (licenseKey.length() > 0) sb.append(" con una clave de licencia (oculta)");
        if (licenseStreamHeaders.length() > 0) sb.append(" y cabeceras propias de licencia");
        sb.append(": el motor de LENY TV solo lo identifica, NO descifra contenido protegido");
        sb.append(" y no se inventan claves. Si la imagen no arranca, es contenido protegido.");
        return sb.toString();
    }

    /** Texto para llevar esta entrada a otra pantalla, sin formularios. */
    String headersSpec() {
        return Headers.toSpec(headers);
    }

    String initial() {
        if (name.length() == 0) return "?";
        return name.substring(0, 1).toUpperCase(java.util.Locale.US);
    }

    @Override
    public String toString() {
        return name + " [" + category + "]";
    }
}

/**
 * Registro de diagnóstico de LENY TV.
 *
 * Guarda paso a paso lo que hace la app (comprobación de internet, cada intento
 * de descarga con su código de respuesta y su tamaño, resultado del análisis de
 * la lista, número de canales que llegan a la pantalla y el error real del
 * reproductor). Sirve para saber EXACTAMENTE en qué punto se interrumpe la
 * carga sin necesidad de conectar el teléfono a un ordenador:
 *
 *  - Se puede ver dentro de la app (botón DIAGNÓSTICO).
 *  - Se puede copiar al portapapeles y pegar en un mensaje.
 *  - Se guarda también en un archivo, así no se pierde si la app se cierra.

final class Diag {

    /** Versión visible del registro: identifica la compilación que lo generó. */
    static final String REVISION = "revisión 55";

    private static final int MAX_LINES = 700;
    private static final int MAX_SAVED_LINES = 700;
    private static final String FILE_NAME = "diagnostico-leny-tv.txt";
    private static final Object LOCK = new Object();
    private static final ArrayList<String> LINES = new ArrayList<String>();
    private static final SimpleDateFormat STAMP = new SimpleDateFormat("HH:mm:ss", Locale.US);
    private static boolean loadedFromFile = false;
    /** Contexto de la app para leer la ficha de instalación (versión y firma). */
    private static Context appContext = null;

    private Diag() {
    }

    /** Añade una línea al registro. Nunca lanza excepciones. */
    static void add(String message) {
        try {
            String clean = message == null ? "" : message.trim();
            if (clean.length() == 0) return;
            if (clean.length() > 600) clean = clean.substring(0, 600) + "…";
            String line = STAMP.format(new Date()) + "  " + clean;
            synchronized (LOCK) {
                LINES.add(line);
                while (LINES.size() > MAX_LINES) LINES.remove(0);
            }
            android.util.Log.i("LenyTV", line);
        } catch (Throwable ignored) {
            // Un registro nunca debe romper la app.
        }
    }

    static void clear() {
        synchronized (LOCK) {
            LINES.clear();
        }
    }

    /** Texto completo del registro, listo para mostrar o copiar. */
    static String text() {
        StringBuilder sb = new StringBuilder();
        sb.append("LENY TV · ").append(REVISION).append('\n');
        sb.append("----------------------------------------\n");
        // Primero el informe del motor: dice qué reproduce cada formato, por dónde
        // pasa el canal y el error técnico completo (códigos crudos, etapa del motor
        // DASH y respuesta real del servidor). Así el motivo se lee de un vistazo,
        // antes del registro paso a paso.
        sb.append(EngineInfo.report());
        sb.append("----------------------------------------\n");
        // Ficha de instalación: identificador, versión, Android mínimo, firma y
        // archivo del que se instaló. Sirve para entender un «App no instalada».
        sb.append(InstallInfo.report(appContext));
        sb.append("----------------------------------------\n");
        sb.append("REGISTRO PASO A PASO\n");
        synchronized (LOCK) {
            if (LINES.isEmpty()) {
                sb.append("(el registro está vacío todavía)");
            } else {
                for (int i = 0; i < LINES.size(); i++) sb.append(LINES.get(i)).append('\n');
            }
        }
        return sb.toString();
    }

    /**
     * Dirección resumida, sin datos que puedan ser privados (usuario, clave o
     * cualquier valor largo de la dirección). Se guarda el servidor y el
     * principio del camino para poder reconocerla.
     */
    static String shortUrl(String url) {
        if (url == null || url.length() == 0) return "(sin enlace)";
        String clean = url.trim();
        try {
            java.net.URL parsed = new java.net.URL(clean);
            String host = parsed.getHost();
            if (host == null || host.length() == 0) return "(enlace local)";
            return parsed.getProtocol() + "://" + host + "/…";
        } catch (Throwable t) {
            if (clean.length() <= 24) return clean;
            return clean.substring(0, 24) + "…";
        }
    }

    /**
     * Dirección de un canal escrita para el registro SIN exponer datos privados.
     * Se conserva el servidor, el puerto, cuántos tramos tiene la ruta y la
     * extensión del enlace, y se añade una marca corta y estable calculada a
     * partir de la dirección completa: así se puede comprobar que la cuadrícula
     * y el reproductor hablan del MISMO enlace (misma marca) sin guardar el
     * usuario, la clave ni el resto de la dirección.
     */
    static String channelUrl(String url) {
        if (url == null || url.trim().length() == 0) return "(sin enlace)";
        String clean = url.trim();
        String mark = mark(clean);
        String extension = extensionOf(clean);
        try {
            java.net.URL parsed = new java.net.URL(clean);
            String host = parsed.getHost();
            if (host == null || host.length() == 0) return "(enlace local) · marca " + mark;
            StringBuilder sb = new StringBuilder();
            sb.append(parsed.getProtocol()).append("://").append(host);
            if (parsed.getPort() > 0) sb.append(':').append(parsed.getPort());
            int segments = 0;
            String path = parsed.getPath() == null ? "" : parsed.getPath();
            for (int i = 0; i < path.length(); i++) {
                if (path.charAt(i) == '/') segments++;
            }
            sb.append(" · ").append(segments).append(" tramos de ruta");
            sb.append(" · ").append(extension.length() == 0 ? "sin extensión" : extension);
            sb.append(" · marca ").append(mark);
            return sb.toString();
        } catch (Throwable t) {
            return extension + " · marca " + mark;
        }
    }

    /** Marca corta y estable de una dirección, calculada sin guardar la dirección. */
    static String mark(String url) {
        if (url == null || url.length() == 0) return "000000";
        int hash = 0x811c9dc5;
        for (int i = 0; i < url.length(); i++) {
            hash ^= url.charAt(i);
            hash *= 0x01000193;
        }
        String hex = Integer.toHexString(hash & 0xFFFFFF);
        while (hex.length() < 6) hex = "0" + hex;
        return hex;
    }

    /** Extensión del enlace (.m3u8, .ts, .mp4...) para saber qué tipo se pidió. */
    static String extensionOf(String url) {
        if (url == null) return "";
        String value = url.trim().toLowerCase(Locale.US);
        int cut = value.length();
        int query = value.indexOf('?');
        if (query >= 0 && query < cut) cut = query;
        int hash = value.indexOf('#');
        if (hash >= 0 && hash < cut) cut = hash;
        value = value.substring(0, cut);
        int slash = value.lastIndexOf('/');
        int dot = value.lastIndexOf('.');
        if (dot < 0 || dot < slash || dot == value.length() - 1) return "";
        String extension = value.substring(dot);
        return extension.length() > 8 ? "" : extension;
    }

    static boolean isEmpty() {
        synchronized (LOCK) {
            return LINES.isEmpty();
        }
    }

    /** Carga la última sesión guardada para no perder el motivo de un cierre. */
    static void init(Context context) {
        if (context != null) {
            try {
                if (appContext == null) appContext = context.getApplicationContext();
            } catch (Throwable ignored) {
            }
        }
        if (context == null || loadedFromFile) return;
        loadedFromFile = true;
        try {
            File file = file(context);
            if (file == null || !file.exists()) return;
            BufferedReader reader = new BufferedReader(new InputStreamReader(new FileInputStream(file), "UTF-8"));
            ArrayList<String> all = new ArrayList<String>();
            String line;
            while ((line = reader.readLine()) != null) all.add(line);
            reader.close();
            int from = Math.max(0, all.size() - 120);
            synchronized (LOCK) {
                for (int i = from; i < all.size(); i++) LINES.add(all.get(i));
            }
        } catch (Throwable ignored) {
        }
    }

    /** Guarda el registro en el archivo de la app. */
    static void save(Context context) {
        if (context == null) return;
        try {
            File file = file(context);
            if (file == null) return;
            String[] lines = text().split("\n");
            StringBuilder sb = new StringBuilder();
            int from = Math.max(0, lines.length - MAX_SAVED_LINES);
            for (int i = from; i < lines.length; i++) sb.append(lines[i]).append('\n');
            FileOutputStream out = new FileOutputStream(file, false);
            OutputStreamWriter writer = new OutputStreamWriter(out, "UTF-8");
            writer.write(sb.toString());
            writer.close();
        } catch (Throwable ignored) {
        }
    }

    private static File file(Context context) {
        try {
            return new File(context.getFilesDir(), FILE_NAME);
        } catch (Throwable t) {
            return null;
        }
    }
}
