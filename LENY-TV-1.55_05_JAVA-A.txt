package com.pocketbuilder.app_wslenytv1792177359057;

import android.content.Context;
import android.graphics.Bitmap;
import android.graphics.BitmapFactory;
import android.util.LruCache;

import java.io.InputStream;

/** Carga las imágenes generadas guardadas en assets/generated. */
final class Art {

    private static final LruCache<String, Bitmap> CACHE = new LruCache<String, Bitmap>(8);

    private Art() {
    }

    static Bitmap generated(Context context, String fileName) {
        Bitmap cached = CACHE.get(fileName);
        if (cached != null) return cached;
        InputStream in = null;
        try {
            in = context.getAssets().open("generated/" + fileName);
            Bitmap bitmap = BitmapFactory.decodeStream(in);
            if (bitmap != null) CACHE.put(fileName, bitmap);
            return bitmap;
        } catch (Exception e) {
            return null;
        } finally {
            if (in != null) {
                try {
                    in.close();
                } catch (Exception ignored) {
                }
            }
        }
    }
}

package com.pocketbuilder.app_wslenytv1792177359057;

import android.content.Context;
import android.view.View;
import android.view.ViewGroup;

/**
 * Fila de controles que se reparte en varias líneas cuando no cabe.
 *
 * El aviso de error del reproductor lleva cinco botones (Reintentar, Elegir
 * canal, Copiar motivo, Probar enlace y Volver). Con una fila normal, en una
 * pantalla estrecha o en vertical los últimos botones se salían de su cuadro y
 * quedaban cortados. Este contenedor coloca cada botón dentro del ancho real
 * disponible y, cuando no cabe, empieza otra línea debajo.
 *
 * No cambia el tamaño de los botones: siguen teniendo el mismo alto táctil.
 */
final class WrapRow extends ViewGroup {

    private final int gapH;
    private final int gapV;

    WrapRow(Context context, int gapHorizontalPx, int gapVerticalPx) {
        super(context);
        this.gapH = Math.max(0, gapHorizontalPx);
        this.gapV = Math.max(0, gapVerticalPx);
    }

    private int contentWidth(int widthMeasureSpec) {
        int size = MeasureSpec.getSize(widthMeasureSpec);
        if (size <= 0) {
            size = getResources().getDisplayMetrics().widthPixels;
        }
        int available = size - getPaddingLeft() - getPaddingRight();
        return Math.max(1, available);
    }

    @Override
    protected void onMeasure(int widthMeasureSpec, int heightMeasureSpec) {
        int available = contentWidth(widthMeasureSpec);
        int x = 0;
        int y = 0;
        int rowHeight = 0;
        int count = getChildCount();
        for (int i = 0; i < count; i++) {
            View child = getChildAt(i);
            if (child == null || child.getVisibility() == View.GONE) continue;
            child.measure(MeasureSpec.makeMeasureSpec(available, MeasureSpec.AT_MOST),
                    MeasureSpec.makeMeasureSpec(0, MeasureSpec.UNSPECIFIED));
            int childWidth = Math.min(child.getMeasuredWidth(), available);
            int childHeight = child.getMeasuredHeight();
            if (x > 0 && x + gapH + childWidth > available) {
                x = 0;
                y += rowHeight + gapV;
                rowHeight = 0;
            }
            x += (x > 0 ? gapH : 0) + childWidth;
            if (childHeight > rowHeight) rowHeight = childHeight;
        }
        int height = getPaddingTop() + getPaddingBottom() + y + rowHeight;
        setMeasuredDimension(MeasureSpec.getSize(widthMeasureSpec),
                resolveSize(height, heightMeasureSpec));
    }

    @Override
    protected void onLayout(boolean changed, int left, int top, int right, int bottom) {
        int available = Math.max(1, right - left - getPaddingLeft() - getPaddingRight());
        int x = 0;
        int y = getPaddingTop();
        int rowHeight = 0;
        int count = getChildCount();
        for (int i = 0; i < count; i++) {
            View child = getChildAt(i);
            if (child == null || child.getVisibility() == View.GONE) continue;
            int childWidth = Math.min(child.getMeasuredWidth(), available);
            int childHeight = child.getMeasuredHeight();
            if (x > 0 && x + gapH + childWidth > available) {
                x = 0;
                y += rowHeight + gapV;
                rowHeight = 0;
            }
            int childLeft = getPaddingLeft() + x + (x > 0 ? gapH : 0);
            child.layout(childLeft, y, childLeft + childWidth, y + childHeight);
            x = childLeft - getPaddingLeft() + childWidth;
            if (childHeight > rowHeight) rowHeight = childHeight;
        }
    }
}

package com.pocketbuilder.app_wslenytv1792177359057;

import android.content.Context;
import android.graphics.Bitmap;
import android.graphics.BitmapFactory;
import android.graphics.drawable.BitmapDrawable;
import android.graphics.drawable.Drawable;
import android.graphics.drawable.GradientDrawable;
import android.view.Gravity;
import android.view.View;

/**
 * Colores y piezas visuales de la pantalla general (barra superior, columna de
 * categorías y cuadrícula), copiadas de la referencia enviada por el usuario.
 *
 * Todo lo de este archivo afecta SOLO a la pantalla de canales: el reproductor y
 * los ajustes conservan su aspecto anterior.
 */
final class RogStyle {

    private RogStyle() {
    }

    /** Fondo de la barra superior: negro con un tono rojizo muy oscuro. */
    static final int BAR_TOP = 0xF20C0709;
    static final int BAR_BOTTOM = 0xE6070405;
    /** Caja azul de la categoría elegida. */
    static final int BLUE_BOX = 0xFF1F5AA6;
    static final int BLUE_BOX_EDGE = 0xFF4A8BE8;
    /** Fondo de la columna de categorías (deja ver el fondo de la pantalla). */
    static final int RAIL_PANEL = 0x8C0A0608;
    static final int RAIL_IDLE = 0x59150E12;
    /** Textos. */
    static final int TEXT = 0xFFF3F1F1;
    static final int TEXT_DIM = 0xFFB4ADAF;
    static final int DATE = 0xFFE4DCDC;
    /** Engranaje verde y botón de apagado de la referencia. */
    static final int GREEN = 0xFF35C46A;
    static final int POWER = 0xFFE0413A;
    /** Recuadro blanco donde va el logo de cada canal. */
    static final int LOGO_BOX = 0xFFFFFFFF;

    /** Pone la imagen de fondo de la referencia; si faltara, un color oscuro. */
    static void applyBackdrop(Context ctx, View view) {
        if (ctx == null || view == null) return;
        try {
            Bitmap bitmap = BitmapFactory.decodeStream(ctx.getAssets().open("generated/fondo_rog.png"));
            if (bitmap != null) {
                BitmapDrawable drawable = new BitmapDrawable(ctx.getResources(), bitmap);
                drawable.setGravity(Gravity.FILL);
                view.setBackground(drawable);
                return;
            }
        } catch (Throwable ignored) {
            // Si la imagen no se puede leer se usa el color de reserva.
        }
        view.setBackgroundColor(0xFF0A0507);
    }

    /** Degradado de la barra superior. */
    static Drawable bar(Context ctx) {
        GradientDrawable g = new GradientDrawable(
                GradientDrawable.Orientation.TOP_BOTTOM, new int[]{BAR_TOP, BAR_BOTTOM});
        g.setCornerRadius(0f);
        return g;
    }

    /** Recuadro blanco del logo de un canal. */
    static Drawable logoBox(Context ctx) {
        GradientDrawable g = new GradientDrawable();
        g.setShape(GradientDrawable.RECTANGLE);
        g.setColor(LOGO_BOX);
        g.setCornerRadius(Ui.dpi(ctx, 3));
        return g;
    }

    /** Fila de la columna de categorías: caja azul cuando está elegida. */
    static Drawable categoryRow(Context ctx, boolean selected) {
        GradientDrawable g = new GradientDrawable();
        g.setShape(GradientDrawable.RECTANGLE);
        g.setColor(selected ? BLUE_BOX : RAIL_IDLE);
        g.setCornerRadius(Ui.dpi(ctx, 2));
        if (selected) g.setStroke(Math.max(1, Ui.dpi(ctx, 1)), BLUE_BOX_EDGE);
        return g;
    }
}
