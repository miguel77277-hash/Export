    private void startPlayback() {
        attemptToken++;
        ui.removeCallbacks(prepareWatchdog);
        releasePlayer();
        hideMessage();
        final Channel channel = currentChannel();
        if (manualType.length() > 0) currentType = manualType;
        else if (typeOrigin.length() == 0) currentType = resolveType(playbackUrl(channel));
        updateTopBar();

        // Qué se envía de verdad y qué protección declara la lista. Los valores
        // sensibles (cookies, tokens) van ocultos tanto aquí como en el aviso.
        Map<String, String> headersForReport = channelHeaders(channel);
        String headerReport = Headers.describe(headersForReport);
        EngineInfo.headers((channel.hasOwnHeaders() ? "del bloque del canal: " : "generales de la app: ")
                + headerReport);
        if (channel.hasOwnHeaders()) {
            Diag.add("Reproductor · bloque del canal: " + channel.blockSummary());
        }
        if (channel.hasLicense()) {
            Diag.add("Reproductor · " + channel.licenseSummary());
            EngineInfo.license(channel.licenseSummary());
        }

        if (FormatDetector.isDash(currentType)) {
            // DASH: lo reproduce SIEMPRE el motor DASH interno de la app.
            // Antes, cuando ese motor fallaba, el mismo manifiesto se entregaba al
            // reproductor del sistema (MediaPlayer), que NO sabe leer DASH: devolvía
            // el código genérico -2147483648 y tapaba el motivo real que ya se había
            // averiguado. Ese paso se ha eliminado.
            EngineInfo.attempt("motor DASH interno de LENY TV (no es ExoPlayer)",
                    currentType, Diag.channelUrl(channel.url));
            EngineInfo.route("RUTA · canal elegido en la cuadrícula → pantalla del reproductor →"
                    + " tipo de contenido detectado " + currentType + routeOrigin()
                    + " → motor DASH interno de LENY TV (lee el manifiesto y pide los trozos)");
            startDash(channel);
            return;
        }

        EngineInfo.attempt("reproductor de Android (android.media.MediaPlayer)",
                currentType, Diag.channelUrl(channel.url));
        EngineInfo.route("RUTA · canal elegido en la cuadrícula → pantalla del reproductor →"
                + " tipo de contenido detectado " + currentType + routeOrigin()
                + " → reproductor de Android (android.media.MediaPlayer)"
                + (useProxy ? " · a través del puente local con cabeceras" : ""));
        Diag.add("Reproductor · " + attemptLabel() + " · canal " + channel.name + " (" + currentType
                + ") · " + Diag.channelUrl(channel.url));
        setBuffering(true);
        try {
            player = new MediaPlayer();
            player.setAudioAttributes(new AudioAttributes.Builder()
                    .setUsage(AudioAttributes.USAGE_MEDIA)
                    .setContentType(AudioAttributes.CONTENT_TYPE_MOVIE)
                    .build());
            player.setOnPreparedListener(new MediaPlayer.OnPreparedListener() {
                @Override
                public void onPrepared(MediaPlayer mp) {
                    ui.removeCallbacks(prepareWatchdog);
                    prepared = true;
                    Diag.add("Reproductor · vídeo preparado en " + attemptLabel() + " ("
                            + mp.getVideoWidth() + "×" + mp.getVideoHeight() + ") · superficie "
                            + (surface != null ? "lista" : "todavía no lista"));
                    videoWidth = mp.getVideoWidth();
                    videoHeight = mp.getVideoHeight();
                    applyAspect();
                    if (surface != null) {
                        try {
                            mp.setSurface(surface);
                        } catch (Exception ignored) {
                        }
                    }
                    mp.start();
                    playing = true;
                    finished = false;
                    setBuffering(false);
                    updatePlayButton();
                    updateTopBar();
                    ui.removeCallbacks(timeTick);
                    ui.postDelayed(timeTick, 1000);
                    scheduleHide();
                }
            });
            player.setOnCompletionListener(new MediaPlayer.OnCompletionListener() {
                @Override
                public void onCompletion(MediaPlayer mp) {
                    ui.post(new Runnable() {
                        @Override
                        public void run() {
                            playing = false;
                            finished = true;
                            setControlsVisible(true);
                            updatePlayButton();
                            timeView.setText("FIN");
                        }
                    });
                }
            });
            player.setOnErrorListener(new MediaPlayer.OnErrorListener() {
                @Override
                public boolean onError(MediaPlayer mp, int what, int extra) {
                    final String detail = describeError(what, extra);
                    // Cuando el código es -2147483648 ("sin detalle") solo los datos que
                    // guarda el propio reproductor explican algo: se recogen aquí.
                    final String metrics = engineMetrics(mp);
                    EngineInfo.rawError("reproductor de Android · what=" + what + ", extra=" + extra
                            + " · " + detail + (metrics.length() == 0 ? "" : " · " + metrics));
                    // El error real del reproductor queda anotado con sus códigos: es lo
                    // que permite distinguir un fallo de la app de uno del servidor,
                    // de la URL, de la autorización o del formato multimedia.
                    Diag.add("Reproductor · ERROR real (what=" + what + ", extra=" + extra + "): " + detail
                            + (metrics.length() == 0 ? "" : " · datos del reproductor: " + metrics)
                            + " · " + attemptLabel() + " · canal " + channel.name
                            + " · " + Diag.channelUrl(channel.url));
                    Diag.save(PlayerActivity.this);
                    ui.post(new Runnable() {
                        @Override
                        public void run() {
                            releasePlayer();
                            setBuffering(false);
                            // Si el primer intento fue directo, se prueba una vez más con el
                            // puente local, que es lo que hace funcionar muchos canales de
                            // proveedor que exigen cabeceras o que redirigen a otro servidor.
                            if (retryWithProxy(channel, "el reproductor devolvió el error " + extra)) {
                                return;
                            }
                            Diag.add("Reproductor · fallo definitivo · " + attemptLabel() + " · canal "
                                    + channel.name + " · " + Diag.channelUrl(channel.url));
                            showMessage("No se pudo reproducir el canal",
                                    detail + "\n\n" + playbackHints()
                                            + "\n\nCabeceras enviadas: " + Headers.describe(channelHeaders(channel))
                                            + (channel.hasLicense()
                                                    ? "\n" + channel.licenseSummary()
                                                            + "\n" + DrmInfo.summary()
                                                    : ""),
                                    true);
                            EngineInfo.headers(Headers.describe(channelHeaders(channel)));
                            if (channel.hasLicense()) EngineInfo.license(channel.licenseSummary());
                            // El código del reproductor casi nunca dice la causa real:
                            // se le pregunta al servidor qué responde ahora mismo para
                            // saber si es 403 (faltan cabeceras), 404 (enlace caducado),
                            // una página web en lugar del vídeo, etc.
                            askServerForReason(channel);
                            Diag.save(PlayerActivity.this);
                        }
                    });
                    return true;
                }
            });
            player.setOnInfoListener(new MediaPlayer.OnInfoListener() {
                @Override
                public boolean onInfo(MediaPlayer mp, int what, int extra) {
                    if (what == MediaPlayer.MEDIA_INFO_BUFFERING_START) {
                        setBuffering(true);
                    } else if (what == MediaPlayer.MEDIA_INFO_BUFFERING_END
                            || what == MediaPlayer.MEDIA_INFO_VIDEO_RENDERING_START) {
                        setBuffering(false);
                    }
                    return false;
                }
            });
            player.setOnVideoSizeChangedListener(new MediaPlayer.OnVideoSizeChangedListener() {
                @Override
                public void onVideoSizeChanged(MediaPlayer mp, int width, int height) {
                    videoWidth = width;
                    videoHeight = height;
                    applyAspect();
                    updateTopBar();
                }
            });

            if (surface != null) player.setSurface(surface);
            Map<String, String> headers = channelHeaders(channel);
            String source = playbackUrl(channel);
            if (useProxy) {
                // Segundo intento: el puente local repite la petición con las cabeceras
                // puestas y sigue las redirecciones, también en cada trozo de vídeo.
                String bridged = LocalProxy.get().proxyUrl(source, headers);
                if (bridged != null) {
                    source = bridged;
                    Diag.add("Reproductor · se reproduce a través del puente local (127.0.0.1),"
                            + " que repite la petición con cabeceras y sigue las redirecciones."
                            + " Cabeceras que repite: " + Headers.names(LocalProxy.bridgeHeaders(headers)));
                } else {
                    useProxy = false;
                    Diag.add("Reproductor · el puente local no se pudo abrir: se vuelve a la conexión directa");
                }
            }
            Diag.add("Reproductor · " + attemptLabel() + " · cabeceras enviadas: "
                    + Net.describeHeaders(headers) + " · " + Diag.channelUrl(source));
            if (surface != null) {
                Diag.add("Reproductor · superficie de vídeo ya lista antes de preparar");
            }
            if (headers == null || headers.isEmpty()) {
                player.setDataSource(this, Uri.parse(source));
            } else {
                player.setDataSource(this, Uri.parse(source), headers);
            }
            player.prepareAsync();
            // Vigilante: si el servidor calla, la app lo dice en pantalla en vez de
            // quedarse con el círculo de carga para siempre.
            watchdogAttempt = attemptToken;
            ui.removeCallbacks(prepareWatchdog);
            ui.postDelayed(prepareWatchdog, PREPARE_TIMEOUT_MS);
        } catch (Throwable t) {
            releasePlayer();
            setBuffering(false);
            EngineInfo.rawError("no se pudo ni preparar el reproductor: " + Net.technical(t));
            showMessage("No se pudo abrir el enlace", describe(t) + "\n\n" + playbackHints(), true);
            askServerForReason(channel);
        }
    }
